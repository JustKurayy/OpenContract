cmake_minimum_required(VERSION 3.24)

project(OpenContract VERSION 0.1.0 LANGUAGES CXX)

option(CONTRACT_BUILD_TESTS "Build project tests" ON)
option(CONTRACT_BUILD_TOOLS "Build command-line inspection tools" ON)
option(CONTRACT_BUILD_LAUNCHER "Build the launcher application" ON)
option(CONTRACT_BUILD_RUNTIME "Build the runtime application" ON)
option(CONTRACT_ENABLE_LOCAL_GAME_TESTS "Enable read-only tests against a local game installation" OFF)
set(CONTRACT_GAME_PATH "" CACHE PATH "Development-only default game installation path")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(ContractWarnings)
include(ContractDependencies)

add_subdirectory(components/core)
add_subdirectory(components/diagnostics)
add_subdirectory(components/datasource)
add_subdirectory(components/filesystem)
add_subdirectory(components/installation)
add_subdirectory(components/binaryio)
add_subdirectory(components/formats)
add_subdirectory(components/assets)
add_subdirectory(components/navigation)
add_subdirectory(components/scene)
add_subdirectory(components/mission)
add_subdirectory(components/modding)
add_subdirectory(components/runtime)
add_subdirectory(components/launcher)
add_subdirectory(components/scripting)

if(CONTRACT_BUILD_TOOLS)
    add_subdirectory(tools/format_inspector)
    add_subdirectory(tools/mod_validator)
    add_subdirectory(apps/inspect)
    add_subdirectory(apps/mod_validate)
endif()

if(CONTRACT_BUILD_RUNTIME)
    add_subdirectory(apps/runtime)
endif()

if(CONTRACT_BUILD_LAUNCHER)
    add_subdirectory(apps/launcher)
endif()

if(CONTRACT_BUILD_TESTS)
    include(CTest)
    enable_testing()
    add_subdirectory(tests)
endif()
